\begin{frame}{Classical Byzantine Agreement}

\begin{center}

\Large

\[
\boxed{\text{Agreement}}
\]

\[
\boxed{\text{Validity}}
\]

\[
\boxed{\text{Termination}}
\]

\end{center}

\pdfpcnote{
Classical Byzantine agreement has three
important properties.\\
\\
Agreement means correct processes
must decide the same value.\\
\\
Validity means the decision must satisfy
a condition related to the proposed values.\\
\\
Termination means correct processes
eventually make a decision.\\
\\
However, validity does not necessarily mean
that the decided value is factually correct.
}

\end{frame}

\node (c) at (7,0) {Correct};

\draw[->,thick] (a) -- (b);
\draw[->,thick] (b) -- (c);

\end{tikzpicture}

\vspace{0.7cm}

\Large

\[
\boxed{n \geq 5t+1}
\]

\end{center}

\vspace{0.2cm}

\begin{itemize}
\item Mobile Byzantine faults
\item Corrupted internal states
\item Agreement maintenance
\end{itemize}

\pdfpcnote{
Bonnet and his coauthors studied
mobile Byzantine agreement.\\
\\
In their model, malicious agents
can move between processes.\\
\\
When an agent leaves a process,
the process becomes cured.\\
\\
A cured process follows the correct algorithm,
but its internal state may still be corrupted.\\
\\
They proved that at least five t plus one
processes are required.\\
\\
Their algorithm also maintains agreement
even when faults continue to move.
}

\end{frame}

% Slide 6: LLM Challenge
% ==========================================

\begin{frame}{A Challenge in Multi-Agent LLMs}

\begin{center}

\Large

\[
[A,A,A,B,B]
\]

\vspace{0.4cm}

\[
\text{Majority}=A
\]

\[
\text{Correct Answer}=B
\]

\vspace{0.4cm}

\begin{alertblock}{Key Observation}
Agreement does not imply answer correctness.
\end{alertblock}

\end{center}

\pdfpcnote{
LLM agents may produce different answers
even when they are functioning normally.\\
\\
For example, three agents may propose A,
while two agents propose B.\\
\\
A majority vote would select A.\\
\\
However, if B is actually correct,
the agents could agree on the wrong answer.\\
\\
Therefore, agreement alone is not enough
to guarantee reliable LLM decisions.
}

\end{frame}

\textbf{Fault-Tolerant Agreement}

\vspace{0.2cm}

$\downarrow$

\vspace{0.2cm}

\[
\boxed{\text{Verified Decision}}
\]

\end{center}

\pdfpcnote{
My current idea is to separate
LLM reasoning from distributed agreement.\\
\\
First, multiple LLM agents generate
candidate solutions.\\
\\
Next, a deterministic verifier checks
whether the candidates satisfy
specific correctness conditions.\\
\\
Finally, distributed processes agree
on a verified candidate.\\
\\
This is still a research idea,
and I need to investigate how it differs
from existing validated Byzantine agreement.
}

\end{frame}
